{{- /* Une intervention de data/conferences.yaml : événement, vidéo chargée au clic, texte, affiche. Paramètres : c (l'entrée), niveau (balise du titre, h2 par défaut), affiche (afficher l'affiche du colloque). */ -}} {{- $c := .c -}} {{- $t := index $c site.Language.Lang -}} {{- $h := .niveau | default "h2" -}} {{- $poster := and .affiche $c.affiche -}} {{- $watch := "" -}} {{- $embed := "" -}} {{- $debut := "" -}} {{- with $c.youtube -}} {{- $watch = printf "https://www.youtube.com/watch?v=%s" . -}} {{- $embed = printf "https://www.youtube-nocookie.com/embed/%s?autoplay=1&rel=0" . -}} {{- with $c.debut -}} {{- $s := int . -}} {{- $watch = printf "%s&t=%ds" $watch $s -}} {{- $embed = printf "%s&start=%d" $embed $s -}} {{- $debut = printf "%d:%02d" (div $s 60) (mod $s 60) -}} {{- if ge $s 3600 }}{{ $debut = printf "%d:%02d:%02d" (div $s 3600) (mod (div $s 60) 60) (mod $s 60) }}{{ end -}} {{- end -}} {{- end -}} {{- $lien := $watch | default $c.url -}}
{{ printf "<%s class=\"video-event\">" $h | safeHTML }}{{ $t.evenement }} {{ $t.lieu }} · {{ time.AsTime $c.date | time.Format ":date_long" }}{{ printf "" $h | safeHTML }}
{{ if $poster }} {{ with resources.Get $c.affiche }} {{ $small := .Resize "440x q85" }} {{ $big := . }} {{ if gt .Width 1400 }}{{ $big = .Resize "1400x q85" }}{{ end }} {{ end }} {{ end }}
